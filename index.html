# Day-22/security_event.sh: classify failed logins by severity
Security_event() {
   echo "Source IP: $1"
   echo "Failed Logins: $2"
   echo "Username: $3"
if [ "$2" -le 5 ]; then  echo "Status: NORMAL"
elif [ "$2" -le 10 ]; then echo "Status: SUSPICIOUS"
else echo "Status: CRITICAL"
fi
}
